## 1. 複雜任務流程設計

### 順序任務執行

```python
from crewai import Agent, Task, Crew, Process

# 創建 Agents
researcher = Agent(
    role="研究員",
    goal="收集和分析市場數據",
    backstory="你是一位經驗豐富的市場研究員，擅長數據分析。",
    verbose=True,
    allow_delegation=False
)

analyst = Agent(
    role="數據分析師",
    goal="分析研究數據並提取洞察",
    backstory="你是數據分析專家，能夠從數據中發現趨勢。",
    verbose=True,
    allow_delegation=False
)

writer = Agent(
    role="報告撰寫員",
    goal="將分析結果撰寫成專業報告",
    backstory="你是專業的商業報告撰寫者。",
    verbose=True,
    allow_delegation=False
)

# 定義任務
research_task = Task(
    description="研究 2024 年 AI 市場趨勢，收集相關數據和報告。",
    agent=researcher,
    expected_output="詳細的市場數據和趨勢總結"
)

analysis_task = Task(
    description="分析研究數據，識別關鍵趨勢和機會。需要使用前一個任務的輸出。",
    agent=analyst,
    expected_output="數據分析報告，包含圖表和洞察",
    context=[research_task]  # 依賴研究任務
)

writing_task = Task(
    description="基於分析結果，撰寫一份專業的市場報告。",
    agent=writer,
    expected_output="完整的市場分析報告",
    context=[research_task, analysis_task]  # 依賴前兩個任務
)

# 創建 Crew（順序執行）
crew = Crew(
    agents=[researcher, analyst, writer],
    tasks=[research_task, analysis_task, writing_task],
    process=Process.sequential,  # 順序執行
    verbose=True
)

# 執行
result = crew.kickoff()
print(result)
```

## 2. 自定義工具集成

### 創建自定義工具

```python
from crewai_tools import BaseTool
from typing import Type
from pydantic import BaseModel, Field

# 定義工具輸入模型
class WebSearchInput(BaseModel):
    """Web 搜索工具的輸入"""
    query: str = Field(..., description="搜索查詢字符串")
    num_results: int = Field(5, description="返回結果數量")

# 創建自定義工具
class WebSearchTool(BaseTool):
    name: str = "網頁搜索"
    description: str = "搜索互聯網獲取最新信息"
    args_schema: Type[BaseModel] = WebSearchInput

    def _run(self, query: str, num_results: int = 5) -> str:
        """執行搜索"""
        # 這裡應該調用真實的搜索 API
        # 為演示使用模擬數據
        results = []
        for i in range(num_results):
            results.append({
                "title": f"關於 {query} 的結果 {i+1}",
                "snippet": f"這是第 {i+1} 條搜索結果的摘要...",
                "url": f"https://example.com/result{i+1}"
            })

        # 格式化輸出
        output = f"搜索 '{query}' 的結果:\n\n"
        for i, result in enumerate(results, 1):
            output += f"{i}. {result['title']}\n"
            output += f"   {result['snippet']}\n"
            output += f"   {result['url']}\n\n"

        return output

# 數據分析工具
class DataAnalysisTool(BaseTool):
    name: str = "數據分析"
    description: str = "分析數值數據並生成統計摘要"

    def _run(self, data: str) -> str:
        """分析數據"""
        # 簡化的數據分析
        try:
            numbers = [float(x.strip()) for x in data.split(',')]
            mean = sum(numbers) / len(numbers)
            maximum = max(numbers)
            minimum = min(numbers)

            return f"""數據分析結果:
            - 樣本數: {len(numbers)}
            - 平均值: {mean:.2f}
            - 最大值: {maximum:.2f}
            - 最小值: {minimum:.2f}
            """
        except Exception as e:
            return f"數據分析失敗: {str(e)}"

# 使用自定義工具
search_tool = WebSearchTool()
analysis_tool = DataAnalysisTool()

researcher_with_tools = Agent(
    role="配備工具的研究員",
    goal="使用各種工具收集和分析信息",
    backstory="你可以使用多種工具來完成研究任務。",
    tools=[search_tool, analysis_tool],
    verbose=True
)

task_with_tools = Task(
    description="搜索 'AI 市場規模' 並分析相關數據。",
    agent=researcher_with_tools,
    expected_output="包含搜索結果和數據分析的報告"
)

crew_with_tools = Crew(
    agents=[researcher_with_tools],
    tasks=[task_with_tools],
    verbose=True
)

result = crew_with_tools.kickoff()
```

## 3. 輸出格式化和驗證

### 結構化輸出

```python
from pydantic import BaseModel, Field
from typing import List

# 定義輸出結構
class MarketInsight(BaseModel):
    """市場洞察"""
    trend: str = Field(..., description="市場趨勢")
    opportunity: str = Field(..., description="商業機會")
    risk: str = Field(..., description="潛在風險")
    recommendation: str = Field(..., description="建議")

class MarketReport(BaseModel):
    """市場報告"""
    title: str = Field(..., description="報告標題")
    summary: str = Field(..., description="執行摘要")
    insights: List[MarketInsight] = Field(..., description="關鍵洞察列表")
    conclusion: str = Field(..., description="結論")

# 創建任務並指定輸出格式
structured_task = Task(
    description="分析 AI 市場並生成結構化報告。",
    agent=analyst,
    expected_output="結構化的市場分析報告",
    output_json=MarketReport  # 指定輸出結構
)

# 創建 Crew
structured_crew = Crew(
    agents=[analyst],
    tasks=[structured_task],
    verbose=True
)

# 執行並獲取結構化結果
result = structured_crew.kickoff()

# 結果是 Pydantic 模型，可以直接訪問屬性
if isinstance(result, MarketReport):
    print(f"標題: {result.title}")
    print(f"摘要: {result.summary}")
    for insight in result.insights:
        print(f"- 趨勢: {insight.trend}")
        print(f"  機會: {insight.opportunity}")
```

### 輸出驗證

```python
from typing import Any

def validate_output(output: Any, expected_structure: dict) -> bool:
    """驗證輸出是否符合預期結構"""
    try:
        # 檢查必需字段
        for field, field_type in expected_structure.items():
            if field not in output:
                print(f"缺少必需字段: {field}")
                return False

            if not isinstance(output[field], field_type):
                print(f"字段 {field} 類型錯誤")
                return False

        return True
    except Exception as e:
        print(f"驗證失敗: {e}")
        return False

# 定義驗證回調
class ValidatedTask(Task):
    """帶驗證的任務"""

    def __init__(self, *args, validation_fn=None, **kwargs):
        super().__init__(*args, **kwargs)
        self.validation_fn = validation_fn

    def execute(self):
        """執行並驗證結果"""
        result = super().execute()

        if self.validation_fn:
            if not self.validation_fn(result):
                raise ValueError("任務輸出未通過驗證")

        return result

# 使用驗證
expected_structure = {
    "title": str,
    "content": str,
    "recommendations": list
}

validated_task = ValidatedTask(
    description="生成市場報告",
    agent=writer,
    expected_output="經過驗證的報告",
    validation_fn=lambda x: validate_output(x, expected_structure)
)
```

### 錯誤處理和重試

```python
from tenacity import retry, stop_after_attempt, wait_exponential

class ResilientCrew(Crew):
    """具有錯誤處理的 Crew"""

    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=4, max=10)
    )
    def kickoff_with_retry(self, inputs=None):
        """帶重試的執行"""
        try:
            return super().kickoff(inputs)
        except Exception as e:
            print(f"執行失敗，正在重試... 錯誤: {e}")
            raise

# 使用
resilient_crew = ResilientCrew(
    agents=[researcher, analyst],
    tasks=[research_task, analysis_task],
    verbose=True
)

try:
    result = resilient_crew.kickoff_with_retry()
except Exception as e:
    print(f"最終失敗: {e}")
    # 實施降級策略
```

### 性能監控

```python
import time
from datetime import datetime

class MonitoredTask(Task):
    """帶監控的任務"""

    def execute(self):
        """執行並記錄性能指標"""
        start_time = time.time()

        try:
            result = super().execute()
            success = True
        except Exception as e:
            result = None
            success = False
            raise
        finally:
            end_time = time.time()
            duration = end_time - start_time

            # 記錄指標
            metrics = {
                "task": self.description,
                "agent": self.agent.role,
                "start_time": datetime.fromtimestamp(start_time),
                "duration": duration,
                "success": success
            }

            print(f"任務指標: {metrics}")
            # 可以發送到監控系統

        return result

# 使用
monitored_task = MonitoredTask(
    description="執行複雜分析",
    agent=analyst,
    expected_output="分析報告"
)
```

### 成本控制

```python
class CostControlledAgent(Agent):
    """成本控制的 Agent"""

    def __init__(self, *args, max_tokens=2000, **kwargs):
        super().__init__(*args, **kwargs)
        self.max_tokens = max_tokens
        self.total_tokens = 0

    def execute_task(self, task):
        """執行任務並追蹤成本"""
        # 檢查是否超過預算
        if self.total_tokens >= self.max_tokens:
            raise ValueError(f"超過 token 預算: {self.total_tokens}/{self.max_tokens}")

        result = super().execute_task(task)

        # 更新使用量（簡化計算）
        estimated_tokens = len(str(result)) // 4
        self.total_tokens += estimated_tokens

        print(f"累計使用 token: {self.total_tokens}/{self.max_tokens}")

        return result

# 使用
cost_controlled_agent = CostControlledAgent(
    role="成本控制的研究員",
    goal="在預算內完成研究",
    backstory="你需要高效利用資源。",
    max_tokens=5000,
    verbose=True
)
```

### 結果緩存

```python
import hashlib
import json
from pathlib import Path

class CachedCrew(Crew):
    """帶緩存的 Crew"""

    def __init__(self, *args, cache_dir="./cache", **kwargs):
        super().__init__(*args, **kwargs)
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(exist_ok=True)

    def _get_cache_key(self, inputs):
        """生成緩存鍵"""
        key_data = {
            "agents": [a.role for a in self.agents],
            "tasks": [t.description for t in self.tasks],
            "inputs": inputs
        }
        key_str = json.dumps(key_data, sort_keys=True)
        return hashlib.md5(key_str.encode()).hexdigest()

    def kickoff(self, inputs=None):
        """帶緩存的執行"""
        cache_key = self._get_cache_key(inputs)
        cache_file = self.cache_dir / f"{cache_key}.json"

        # 檢查緩存
        if cache_file.exists():
            print(f"使用緩存結果: {cache_key}")
            with open(cache_file, 'r') as f:
                return json.load(f)

        # 執行任務
        result = super().kickoff(inputs)

        # 保存到緩存
        with open(cache_file, 'w') as f:
            json.dump(result, f)

        return result
```

## 6. 實際部署案例

### FastAPI 集成

```python
from fastapi import FastAPI, BackgroundTasks
from pydantic import BaseModel

app = FastAPI()

class ResearchRequest(BaseModel):
    topic: str
    depth: str = "basic"

class ResearchResponse(BaseModel):
    task_id: str
    status: str

# 任務隊列（實際應使用 Redis/Celery）
tasks_status = {}

@app.post("/research", response_model=ResearchResponse)
async def start_research(request: ResearchRequest, background_tasks: BackgroundTasks):
    """啟動研究任務"""
    import uuid
    task_id = str(uuid.uuid4())

    tasks_status[task_id] = {"status": "pending"}

    # 在後台執行
    background_tasks.add_task(run_research_crew, task_id, request.topic, request.depth)

    return ResearchResponse(task_id=task_id, status="started")

@app.get("/research/{task_id}")
async def get_research_status(task_id: str):
    """獲取任務狀態"""
    if task_id not in tasks_status:
        return {"error": "Task not found"}

    return tasks_status[task_id]

def run_research_crew(task_id: str, topic: str, depth: str):
    """執行研究 Crew"""
    try:
        tasks_status[task_id]["status"] = "running"

        # 創建並執行 Crew
        crew = Crew(
            agents=[researcher, analyst, writer],
            tasks=[research_task, analysis_task, writing_task],
            verbose=True
        )

        result = crew.kickoff()

        tasks_status[task_id] = {
            "status": "completed",
            "result": str(result)
        }
    except Exception as e:
        tasks_status[task_id] = {
            "status": "failed",
            "error": str(e)
        }
```
