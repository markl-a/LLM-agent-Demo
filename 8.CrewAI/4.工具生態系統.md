## 工具基礎

### 什麼是工具？

工具是 Agent 可以調用的函數或方法,用於執行特定操作:

```python
from crewai import Agent
from crewai_tools import SerperDevTool

# 創建搜索工具
search_tool = SerperDevTool()

# 為 Agent 配備工具
agent = Agent(
    role='研究員',
    goal='研究最新技術趨勢',
    backstory='你是專業的技術研究員',
    tools=[search_tool],  # 添加工具
    verbose=True
)
```

### 工具架構

```
┌─────────────────────────────────────┐
│            Agent                    │
│  ┌──────────────────────────────┐   │
│  │    Tool Manager              │   │
│  │  ┌────────┐  ┌────────┐     │   │
│  │  │ Tool 1 │  │ Tool 2 │ ... │   │
│  │  └────────┘  └────────┘     │   │
│  └──────────────────────────────┘   │
│              │                       │
│              ▼                       │
│      ┌──────────────┐               │
│      │ LLM Decision │               │
│      └──────────────┘               │
└─────────────────────────────────────┘
         │
         ▼
   Tool Execution
```

## 自定義工具

創建符合特定需求的自定義工具。

### 基礎自定義工具

```python
from crewai_tools import BaseTool
from typing import Type
from pydantic import BaseModel, Field

# 定義工具輸入
class CalculatorInput(BaseModel):
    """計算器輸入"""
    operation: str = Field(..., description="操作類型: add, subtract, multiply, divide")
    a: float = Field(..., description="第一個數字")
    b: float = Field(..., description="第二個數字")

# 創建自定義工具
class CalculatorTool(BaseTool):
    name: str = "計算器"
    description: str = "執行基本數學運算"
    args_schema: Type[BaseModel] = CalculatorInput

    def _run(self, operation: str, a: float, b: float) -> str:
        """執行計算"""
        operations = {
            'add': lambda x, y: x + y,
            'subtract': lambda x, y: x - y,
            'multiply': lambda x, y: x * y,
            'divide': lambda x, y: x / y if y != 0 else "除數不能為零"
        }

        if operation not in operations:
            return f"不支持的操作: {operation}"

        result = operations[operation](a, b)
        return f"{a} {operation} {b} = {result}"

# 使用自定義工具
calculator = CalculatorTool()

math_agent = Agent(
    role='數學助手',
    goal='幫助用戶進行數學計算',
    tools=[calculator],
    verbose=True
)
```

#### 1. 天氣查詢工具

from crewai_tools import BaseTool
from pydantic import BaseModel, Field
import requests

class WeatherInput(BaseModel):

class WeatherTool(BaseTool):
    name: str = "天氣查詢"
    description: str = "查詢指定城市的天氣信息"
    args_schema: Type[BaseModel] = WeatherInput

    def _run(self, city: str) -> str:
        """查詢天氣"""
        # 這裡使用實際的天氣 API
        # 示例使用模擬數據
        weather_data = {
            "台北": {"temp": 25, "condition": "晴天"},
            "高雄": {"temp": 28, "condition": "多雲"},
            "台中": {"temp": 26, "condition": "陰天"}
        }

        if city in weather_data:
            data = weather_data[city]
            return f"{city}的天氣：{data['condition']}，溫度：{data['temp']}°C"
        else:
            return f"找不到 {city} 的天氣信息"

weather_tool = WeatherTool()

#### 2. 數據庫查詢工具

```python
from crewai_tools import BaseTool
from pydantic import BaseModel, Field
import sqlite3

class DatabaseQueryInput(BaseModel):
    query: str = Field(..., description="SQL 查詢語句")

class DatabaseTool(BaseTool):
    name: str = "數據庫查詢"
    description: str = "執行 SQL 查詢並返回結果"
    args_schema: Type[BaseModel] = DatabaseQueryInput

    def __init__(self, db_path: str, **kwargs):
        super().__init__(**kwargs)
        self.db_path = db_path

    def _run(self, query: str) -> str:
        """執行數據庫查詢"""
        try:
            conn = sqlite3.connect(self.db_path)
            cursor = conn.cursor()
            cursor.execute(query)
            results = cursor.fetchall()
            conn.close()

            if results:
                return f"查詢結果：{results}"
            else:
                return "查詢沒有返回結果"
        except Exception as e:
            return f"查詢錯誤：{str(e)}"

db_tool = DatabaseTool(db_path='./data/company.db')
```

#### 3. API 調用工具

```python
from crewai_tools import BaseTool
from pydantic import BaseModel, Field
import requests

class APICallInput(BaseModel):
    endpoint: str = Field(..., description="API 端點")
    method: str = Field("GET", description="HTTP 方法")
    params: dict = Field({}, description="查詢參數")

class APITool(BaseTool):
    name: str = "API 調用工具"
    description: str = "調用 REST API 並返回結果"
    args_schema: Type[BaseModel] = APICallInput

    def __init__(self, base_url: str, api_key: str = None, **kwargs):
        super().__init__(**kwargs)
        self.base_url = base_url
        self.api_key = api_key

    def _run(self, endpoint: str, method: str = "GET", params: dict = {}) -> str:
        """執行 API 調用"""
        url = f"{self.base_url}/{endpoint}"
        headers = {}

        if self.api_key:
            headers['Authorization'] = f'Bearer {self.api_key}'

        try:
            if method == "GET":
                response = requests.get(url, params=params, headers=headers)
            elif method == "POST":
                response = requests.post(url, json=params, headers=headers)
            else:
                return f"不支持的 HTTP 方法: {method}"

            response.raise_for_status()
            return response.json()
        except Exception as e:
            return f"API 調用失敗: {str(e)}"

api_tool = APITool(
    base_url='https://api.example.com',
    api_key='your-api-key'
)
```

#### 4. 郵件發送工具

```python
from crewai_tools import BaseTool
from pydantic import BaseModel, Field
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart

class EmailInput(BaseModel):
    to: str = Field(..., description="收件人郵箱")
    subject: str = Field(..., description="郵件主題")
    body: str = Field(..., description="郵件內容")

class EmailTool(BaseTool):
    name: str = "郵件發送工具"
    description: str = "發送電子郵件"
    args_schema: Type[BaseModel] = EmailInput

    def __init__(self, smtp_server: str, smtp_port: int, username: str, password: str, **kwargs):
        super().__init__(**kwargs)
        self.smtp_server = smtp_server
        self.smtp_port = smtp_port
        self.username = username
        self.password = password

    def _run(self, to: str, subject: str, body: str) -> str:
        """發送郵件"""
        try:
            msg = MIMEMultipart()
            msg['From'] = self.username
            msg['To'] = to
            msg['Subject'] = subject
            msg.attach(MIMEText(body, 'plain'))

            server = smtplib.SMTP(self.smtp_server, self.smtp_port)
            server.starttls()
            server.login(self.username, self.password)
            server.send_message(msg)
            server.quit()

            return f"郵件已成功發送到 {to}"
        except Exception as e:
            return f"郵件發送失敗: {str(e)}"

# 使用
email_tool = EmailTool(
    smtp_server='smtp.gmail.com',
    smtp_port=587,
    username='your-email@gmail.com',
    password='your-password'
)
```

## 最佳實踐
