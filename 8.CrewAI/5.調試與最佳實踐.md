```python
def cleanup_old_memories(crew_id, days=30):
    """清理舊記憶"""
```

```python
from tenacity import retry, stop_after_attempt, wait_exponential

class ResilientCrew(Crew):
```

```python
def batch_process_tasks(crew, task_inputs, batch_size=5):
    """批量處理任務"""
    results = []

    for i in range(0, len(task_inputs), batch_size):
        batch = task_inputs[i:i+batch_size]
        print(f"處理批次 {i//batch_size + 1}")

        batch_results = []
        for task_input in batch:
            result = crew.kickoff(task_input)
            batch_results.append(result)

        results.extend(batch_results)

        # 批次間延遲,避免 API 限流
        import time
        time.sleep(1)

    return results
```

### 1. Token 使用監控

```python
import tiktoken

class TokenCounter:
    """Token 計數器"""

    def __init__(self, model="gpt-4"):
        self.encoding = tiktoken.encoding_for_model(model)
        self.total_tokens = 0

    def count_tokens(self, text: str) -> int:
        """計算 token 數量"""
        tokens = len(self.encoding.encode(text))
        self.total_tokens += tokens
        return tokens

    def estimate_cost(self, input_tokens: int, output_tokens: int, model="gpt-4") -> float:
        """估算成本"""
        costs = {
            "gpt-4": {"input": 0.03 / 1000, "output": 0.06 / 1000},
            "gpt-3.5-turbo": {"input": 0.001 / 1000, "output": 0.002 / 1000}
        }

        if model in costs:
            cost = (input_tokens * costs[model]["input"] +
                   output_tokens * costs[model]["output"])
            return cost
        return 0

# 使用
counter = TokenCounter()
tokens = counter.count_tokens("這是測試文本")
cost = counter.estimate_cost(1000, 500, "gpt-4")
print(f"預估成本: ${cost:.4f}")
```

```python
from fastapi import FastAPI, BackgroundTasks, HTTPException
from pydantic import BaseModel
from typing import Dict
import uuid

app = FastAPI()

# 任務狀態存儲
task_results: Dict[str, dict] = {}

class TaskRequest(BaseModel):
    input: str
    config: dict = {}

class TaskResponse(BaseModel):
    task_id: str
    status: str

@app.post("/tasks", response_model=TaskResponse)
async def create_task(request: TaskRequest, background_tasks: BackgroundTasks):
    """創建新任務"""
    task_id = str(uuid.uuid4())

    # 初始化任務狀態
    task_results[task_id] = {
        "status": "pending",
        "result": None
    }

    # 在背景執行
    background_tasks.add_task(run_crew_task, task_id, request.input)

    return TaskResponse(task_id=task_id, status="started")

@app.get("/tasks/{task_id}")
async def get_task_status(task_id: str):
    """獲取任務狀態"""
    if task_id not in task_results:
        raise HTTPException(status_code=404, detail="任務不存在")

    return task_results[task_id]

def run_crew_task(task_id: str, input_text: str):
    """執行 Crew 任務"""
    try:
        task_results[task_id]["status"] = "running"

        # 執行 CrewAI 任務
        from crewai import Agent, Task, Crew

        agent = Agent(
            role='處理器',
            goal='處理用戶輸入',
            backstory='你是專業的處理專家',
            verbose=False
        )

        task = Task(
            description=input_text,
            agent=agent,
            expected_output='處理結果'
        )

        crew = Crew(agents=[agent], tasks=[task], verbose=False)
        result = crew.kickoff()

        task_results[task_id] = {
            "status": "completed",
            "result": str(result)
        }
    except Exception as e:
        task_results[task_id] = {
            "status": "failed",
            "error": str(e)
        }

# 運行: uvicorn main:app --host 0.0.0.0 --port 8000
```

```python
import time
from functools import wraps

def monitor_performance(func):
    """性能監控裝飾器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start_time = time.time()
        start_memory = get_memory_usage()

        try:
            result = func(*args, **kwargs)
            success = True
        except Exception as e:
            result = None
            success = False
            raise
        finally:
            end_time = time.time()
            end_memory = get_memory_usage()

            metrics = {
                "function": func.__name__,
                "duration": end_time - start_time,
                "memory_delta": end_memory - start_memory,
                "success": success,
                "timestamp": datetime.now().isoformat()
            }

            log_metrics(metrics)

        return result
    return wrapper

def get_memory_usage():
    """獲取內存使用量"""
    import psutil
    process = psutil.Process()
    return process.memory_info().rss / 1024 / 1024  # MB

def log_metrics(metrics):
    """記錄指標"""
    print(f"性能指標: {metrics}")

# 使用
@monitor_performance
def run_crew_task(crew):
    return crew.kickoff()
```

```python
import os
from cryptography.fernet import Fernet

class SecretManager:
    """密鑰管理器"""

    def __init__(self):
        # 從環境變數讀取加密密鑰
        key = os.getenv('ENCRYPTION_KEY')
        if not key:
            key = Fernet.generate_key()
        self.cipher = Fernet(key)

    def encrypt(self, value: str) -> str:
        """加密"""
        return self.cipher.encrypt(value.encode()).decode()

    def decrypt(self, encrypted_value: str) -> str:
        """解密"""
        return self.cipher.decrypt(encrypted_value.encode()).decode()

# 使用
secret_manager = SecretManager()
encrypted_key = secret_manager.encrypt(os.getenv('OPENAI_API_KEY'))
```

```python
from pydantic import BaseModel, Field, validator

class ValidatedInput(BaseModel):
    """驗證的輸入"""
    query: str = Field(..., min_length=1, max_length=1000)
    user_id: str = Field(..., regex=r'^[a-zA-Z0-9_-]+$')
```

```python
import re

def sanitize_output(text: str) -> str:
    """清理輸出"""
    # 移除個人信息
    patterns = {
        'email': r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b',
        'phone': r'\b\d{3}[-.]?\d{3}[-.]?\d{4}\b',
        'ssn': r'\b\d{3}-\d{2}-\d{4}\b',
        'credit_card': r'\b\d{4}[-\s]?\d{4}[-\s]?\d{4}[-\s]?\d{4}\b'
    }

    sanitized = text
    for pattern_type, pattern in patterns.items():
        sanitized = re.sub(pattern, f'[REDACTED_{pattern_type.upper()}]', sanitized)

    return sanitized

# 使用
output = "聯繫我: email@example.com, 電話 123-456-7890"
safe_output = sanitize_output(output)
print(safe_output)
```

### 4. 速率限制

```python
from datetime import datetime, timedelta
from collections import defaultdict

class RateLimiter:
    """速率限制器"""

    def __init__(self, max_requests=10, time_window=60):
        self.max_requests = max_requests
        self.time_window = timedelta(seconds=time_window)
        self.requests = defaultdict(list)

    def is_allowed(self, user_id: str) -> bool:
        """檢查是否允許請求"""
        now = datetime.now()
        cutoff = now - self.time_window

        # 清理過期請求
        self.requests[user_id] = [
            req_time for req_time in self.requests[user_id]
            if req_time > cutoff
        ]

        # 檢查是否超限
        if len(self.requests[user_id]) >= self.max_requests:
            return False

        # 記錄請求
        self.requests[user_id].append(now)
        return True

# 使用
limiter = RateLimiter(max_requests=5, time_window=60)

if limiter.is_allowed("user_123"):
    print("請求允許")
else:
    print("速率限制")
```
