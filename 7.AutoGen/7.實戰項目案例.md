## 案例 1：智能數據分析助手

### 項目概述

**目標**：構建一個能夠自動分析數據、生成可視化圖表和提供業務洞察的智能助手。

**應用場景**：
- 銷售數據分析
- 用戶行為分析
- 財務報表分析
- 市場趨勢研究

### 系統架構

```
用戶輸入
    ↓
[數據理解Agent] → 理解需求和數據結構
    ↓
[數據清洗Agent] → 處理缺失值、異常值
    ↓
[統計分析Agent] → 計算統計指標
    ↓
[可視化Agent] → 生成圖表
    ↓
[洞察生成Agent] → 提供業務建議
    ↓
分析報告
```

### 完整實現

```python

from typing import List, Dict

        self.executor.initiate_chat(
```

```python
# 生成示例銷售數據

        "產品": np.random.choice(products, n_rows),
        "區域": np.random.choice(regions, n_rows),
        "銷售額": np.random.normal(5000, 2000, n_rows).round(2),
        "數量": np.random.poisson(50, n_rows),
        "折扣": np.random.uniform(0, 0.3, n_rows).round(2),
    }

    df = pd.DataFrame(data)
    df.to_csv("sales_data.csv", index=False)
    print("示例數據已生成：sales_data.csv")
```

### 部署建議

```python
# app.py - Web 應用示例
from flask import Flask, request, jsonify
import threading

app = Flask(__name__)
team = DataAnalysisTeam()

@app.route('/analyze', methods=['POST'])
def analyze_data():
    """分析數據 API"""
    data = request.json
    data_path = data.get('data_path')
    goal = data.get('goal')

    # 異步執行分析
    thread = threading.Thread(
        target=team.analyze,
        args=(data_path, goal)
    )
    thread.start()

    return jsonify({
        "status": "分析已啟動",
        "message": "分析結果將保存到 workspace 目錄"
    })

if __name__ == '__main__':
    app.run(debug=True, port=5000)
```

class CodeReviewSystem:

            groupchat=groupchat,
            llm_config=llm_config

## 案例 3：多語言客服機器人

### 項目概述

**目標**：構建一個支持多語言、能夠處理複雜問題並自動升級的智能客服系統。

### 完整實現

```python
class MultilingualCustomerService:
    """多語言客服系統"""

    def __init__(self):
        # L1 客服（處理常見問題）
        self.l1_support = autogen.AssistantAgent(
            name="L1客服",
            system_message="""你是一線客服人員。

            職責：
            - 友好、專業的服務態度
            - 處理常見問題（帳號、密碼、基本操作）
            - 提供快速解決方案
            - 支持多語言（中文、英文）

            如果無法解決，回復："需要升級到技術支持 [ESCALATE]"

            回復模板：
            【問題理解】...
            【解決方案】...
            【後續建議】...
            """,
            llm_config=llm_config
        )

        # L2 技術支持
        self.l2_support = autogen.AssistantAgent(
            name="技術支持",
            system_message="""你是技術支持專家。

            職責：
            - 處理技術問題
            - 診斷系統故障
            - 提供詳細解決步驟
            - 必要時提供代碼示例

            如果需要工程師介入，回復："需要工程師處理 [ESCALATE]"
            """,
            llm_config=llm_config
        )

        # 工程師
        self.engineer = autogen.AssistantAgent(
            name="工程師",
            system_message="""你是後端工程師。

            職責：
            - 解決代碼層面的問題
            - 修復系統Bug
            - 性能優化
            - 提供技術方案

            提供詳細的技術分析和解決方案。
            """,
            llm_config=llm_config
        )

        # 用戶代理
        self.user_proxy = autogen.UserProxyAgent(
            name="用戶",
            human_input_mode="NEVER",
            max_consecutive_auto_reply=10,
            code_execution_config=False
        )

        # 知識庫
        self.knowledge_base = {
            "重置密碼": "請訪問 www.example.com/reset 重置密碼",
            "綁定郵箱": "在設置 -> 帳號安全 -> 綁定郵箱",
            "API限流": "預設每分鐘100次請求，可聯繫升級",
        }

    def handle_question(self, question: str, language: str = "zh"):
        """處理客戶問題"""
        # 檢查知識庫
        for keyword, answer in self.knowledge_base.items():
            if keyword in question:
                print(f"📚 知識庫匹配：{answer}")
                return answer

        # 創建升級邏輯
        def escalation_logic(last_speaker, groupchat):
            """客服升級邏輯"""
            if not groupchat.messages:
                return self.l1_support

            last_message = groupchat.messages[-1]["content"]

            # 檢查是否需要升級
            if "[ESCALATE]" in last_message:
                if last_speaker == self.l1_support:
                    print("📈 升級到 L2 技術支持")
                    return self.l2_support
                elif last_speaker == self.l2_support:
                    print("📈 升級到工程師")
                    return self.engineer

            # 默認流程
            if last_speaker == self.user_proxy:
                return self.l1_support

            return self.user_proxy

        # 創建群組
        groupchat = autogen.GroupChat(
            agents=[
                self.user_proxy,
                self.l1_support,
                self.l2_support,
                self.engineer
            ],
            messages=[],
            max_round=15,
            speaker_selection_method=escalation_logic
        )

        manager = autogen.GroupChatManager(
            groupchat=groupchat,
            llm_config=llm_config
        )

        # 開始對話
        self.user_proxy.initiate_chat(
            manager,
            message=f"[{language}] {question}"
        )

# 使用示例
service = MultilingualCustomerService()

# 測試問題
questions = [
    "我忘記密碼了，怎麼辦？",  # L1 可以處理
    "API 調用返回 429 錯誤，如何解決？",  # 需要 L2
    "系統偶爾出現記憶體洩漏，如何排查？",  # 需要工程師
]

for q in questions:
    print(f"\n{'='*60}")
    print(f"問題：{q}")
    print('='*60)
    service.handle_question(q)
```

```python
        """完成任務"""
        for task in self.tasks:
            if task["id"] == task_id:
                task["status"] = "完成"
                task["completed_at"] = datetime.now()
                self.completed_tasks.append(task)
                break

    def get_report(self):
        """獲取項目報告"""
        total = len(self.tasks)
        completed = len(self.completed_tasks)
        return {
            "總任務數": total,
            "已完成": completed,
            "完成率": f"{(completed/total*100):.1f}%" if total > 0 else "0%"
        }
```

```python
class AgentError(Exception):
    """Agent 錯誤基類"""
    pass

def safe_chat(agent, message, max_retries=3):
    """安全的對話執行"""
    for attempt in range(max_retries):
        try:
            return agent.initiate_chat(assistant, message=message)
        except Exception as e:
            logger.error(f"對話失敗 (嘗試 {attempt + 1}/{max_retries}): {e}")
            if attempt == max_retries - 1:
                raise AgentError(f"對話失敗: {e}")
            time.sleep(2 ** attempt)
```
