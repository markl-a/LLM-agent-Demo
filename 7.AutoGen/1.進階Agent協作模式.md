## 1. 群組聊天 (GroupChat)

群組聊天允許多個 Agent 在一個對話中協作，自動管理發言順序。

### 基本群組聊天

```python
import autogen

# 配置 LLM
config_list = [
    {
        "model": "gpt-4",
        "api_key": "your-api-key"
    }
]

llm_config = {"config_list": config_list}

# 創建 Agent
user_proxy = autogen.UserProxyAgent(
    name="User",
    human_input_mode="NEVER",
    code_execution_config={"use_docker": False}
)

researcher = autogen.AssistantAgent(
    name="Researcher",
    system_message="你是一個研究員，負責收集和分析信息。",
    llm_config=llm_config
)

writer = autogen.AssistantAgent(
    name="Writer",
    system_message="你是一個作家，負責將研究結果撰寫成文章。",
    llm_config=llm_config
)

critic = autogen.AssistantAgent(
    name="Critic",
    system_message="你是一個評論家，負責審查和改進內容。",
    llm_config=llm_config
)

# 創建群組聊天
groupchat = autogen.GroupChat(
    agents=[user_proxy, researcher, writer, critic],
    messages=[],
    max_round=10
)

# 創建群組聊天管理器
manager = autogen.GroupChatManager(
    groupchat=groupchat,
    llm_config=llm_config
)

# 啟動對話
user_proxy.initiate_chat(
    manager,
    message="請撰寫一篇關於 AI 在醫療領域應用的文章。"
)
```

### 自定義發言順序

```python
# 方法1：輪流發言
def round_robin_speaker_selection(last_speaker, groupchat):
    """輪流選擇下一個發言者"""
    agents = groupchat.agents
    current_idx = agents.index(last_speaker)
    next_idx = (current_idx + 1) % len(agents)
    return agents[next_idx]

groupchat = autogen.GroupChat(
    agents=[user_proxy, researcher, writer, critic],
    messages=[],
    max_round=10,
    speaker_selection_method=round_robin_speaker_selection
)

# 方法2：基於角色的選擇
def role_based_selection(last_speaker, groupchat):
    """根據上一個發言者決定下一個"""
    if last_speaker.name == "Researcher":
        return next(a for a in groupchat.agents if a.name == "Writer")
    elif last_speaker.name == "Writer":
        return next(a for a in groupchat.agents if a.name == "Critic")
    elif last_speaker.name == "Critic":
        return next(a for a in groupchat.agents if a.name == "User")
    else:
        return next(a for a in groupchat.agents if a.name == "Researcher")

groupchat = autogen.GroupChat(
    agents=[user_proxy, researcher, writer, critic],
    messages=[],
    max_round=10,
    speaker_selection_method=role_based_selection
)
```

## 2. 嵌套對話 (Nested Chats)

嵌套對話允許在主對話中觸發子對話，實現更複雜的協作模式。

### 基本嵌套對話

```python
# 創建主 Agent
project_manager = autogen.AssistantAgent(
    name="ProjectManager",
    system_message="你是項目經理，負責協調團隊完成任務。",
    llm_config=llm_config
)

# 創建子任務 Agents
backend_dev = autogen.AssistantAgent(
    name="BackendDev",
    system_message="你是後端開發工程師。",
    llm_config=llm_config
)

frontend_dev = autogen.AssistantAgent(
    name="FrontendDev",
    system_message="你是前端開發工程師。",
    llm_config=llm_config
)

# 定義嵌套對話觸發函數
def trigger_backend_chat(recipient, messages, sender, config):
    """觸發後端開發對話"""
    return backend_dev.generate_reply(
        messages=messages,
        sender=sender
    )

def trigger_frontend_chat(recipient, messages, sender, config):
    """觸發前端開發對話"""
    return frontend_dev.generate_reply(
        messages=messages,
        sender=sender
    )

# 註冊嵌套對話
project_manager.register_nested_chats(
    [
        {
            "chat_queue": [backend_dev],
            "summary_method": "last_msg",
            "trigger": lambda sender: "backend" in sender.last_message()["content"].lower()
        },
        {
            "chat_queue": [frontend_dev],
            "summary_method": "last_msg",
            "trigger": lambda sender: "frontend" in sender.last_message()["content"].lower()
        }
    ]
)

# 啟動對話
user_proxy.initiate_chat(
    project_manager,
    message="我們需要開發一個 Web 應用，包含後端 API 和前端界面。"
)
```

## 3. 工具和函數調用

讓 Agent 能夠調用外部工具和函數。

### 註冊自定義函數

```python
from typing import Annotated, Literal

# 定義工具函數
def search_web(query: Annotated[str, "搜索查詢"]) -> str:
    """
    搜索網絡並返回結果

    Args:
        query: 搜索查詢字符串

    Returns:
        搜索結果
    """
    # 模擬搜索
    return f"關於 '{query}' 的搜索結果：這是一些相關信息..."

def calculate(
    operation: Annotated[Literal["add", "subtract", "multiply", "divide"], "運算類型"],
    a: Annotated[float, "第一個數字"],
    b: Annotated[float, "第二個數字"]
) -> float:
    """
    執行數學運算

    Args:
        operation: 運算類型
        a: 第一個數字
        b: 第二個數字

    Returns:
        運算結果
    """
    if operation == "add":
        return a + b
    elif operation == "subtract":
        return a - b
    elif operation == "multiply":
        return a * b
    elif operation == "divide":
        return a / b if b != 0 else float('inf')

# 創建支持函數調用的 Agent
assistant = autogen.AssistantAgent(
    name="Assistant",
    llm_config={
        "config_list": config_list,
        "functions": [
            {
                "name": "search_web",
                "description": "搜索網絡獲取信息",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "query": {
                            "type": "string",
                            "description": "搜索查詢"
                        }
                    },
                    "required": ["query"]
                }
            },
            {
                "name": "calculate",
                "description": "執行數學運算",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "operation": {
                            "type": "string",
                            "enum": ["add", "subtract", "multiply", "divide"]
                        },
                        "a": {"type": "number"},
                        "b": {"type": "number"}
                    },
                    "required": ["operation", "a", "b"]
                }
            }
        ]
    }
)

# 註冊函數執行
user_proxy = autogen.UserProxyAgent(
    name="User",
    human_input_mode="NEVER",
    function_map={
        "search_web": search_web,
        "calculate": calculate
    }
)

# 啟動對話
user_proxy.initiate_chat(
    assistant,
    message="搜索 'AutoGen framework' 並計算 25 乘以 4。"
)
```

## 4. 狀態管理和持久化

### 保存和加載對話歷史

```python
import json

# 保存對話
def save_conversation(agent, filename):
    """保存對話歷史"""
    messages = agent.chat_messages
    with open(filename, 'w', encoding='utf-8') as f:
        json.dump(messages, f, ensure_ascii=False, indent=2)

# 加載對話
def load_conversation(agent, filename):
    """加載對話歷史"""
    with open(filename, 'r', encoding='utf-8') as f:
        messages = json.load(f)
    agent.chat_messages = messages

# 使用
user_proxy.initiate_chat(assistant, message="你好！")
save_conversation(user_proxy, "conversation.json")

# 稍後恢復
load_conversation(user_proxy, "conversation.json")
user_proxy.initiate_chat(assistant, message="繼續之前的對話...")
```

### 自定義狀態追蹤

```python
class StatefulAgent(autogen.AssistantAgent):
    """帶狀態追蹤的 Agent"""

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.state = {
            "tasks_completed": [],
            "current_task": None,
            "context": {}
        }

    def update_state(self, key, value):
        """更新狀態"""
        self.state[key] = value

    def get_state(self, key):
        """獲取狀態"""
        return self.state.get(key)

    def generate_reply(self, messages=None, sender=None, **kwargs):
        """生成回復時更新狀態"""
        reply = super().generate_reply(messages, sender, **kwargs)

        # 更新狀態邏輯
        if "completed" in reply.get("content", "").lower():
            self.state["tasks_completed"].append(self.state["current_task"])

        return reply

# 使用
stateful_agent = StatefulAgent(
    name="StatefulAssistant",
    llm_config=llm_config
)

# 檢查狀態
print(stateful_agent.get_state("tasks_completed"))
```

def safe_chat(agent, message, max_retries=3):
    """帶重試的安全對話"""
    for attempt in range(max_retries):
        try:
            return user_proxy.initiate_chat(agent, message=message)
        except Exception as e:
            print(f"嘗試 {attempt + 1} 失敗: {e}")
