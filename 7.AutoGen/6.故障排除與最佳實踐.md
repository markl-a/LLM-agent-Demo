## 常見問題與解決方案

### 問題 1：Agent 對話陷入無限循環

**症狀**：
- Agent 不斷重複相同的對話
- 達到 `max_consecutive_auto_reply` 限制
- API 成本快速增長

**原因**：
- 沒有明確的終止條件
- Agent 互相等待對方結束
- system_message 不清晰導致角色混淆

**解決方案**：

```python
# ✅ 方案 1：設置明確的終止條件
agent = autogen.AssistantAgent(
    name="assistant",
    llm_config=llm_config,
    is_termination_msg=lambda x: x.get("content", "").rstrip().endswith("TERMINATE"),
    max_consecutive_auto_reply=10  # 設置上限
)

# ✅ 方案 2：在 system_message 中說明何時終止
system_message = """你是一個助手。
完成任務後，請在回復末尾添加 'TERMINATE'。

終止條件：
1. 問題已完全解決
2. 用戶明確表示滿意
3. 達到目標狀態
"""

# ✅ 方案 3：使用複雜的終止邏輯
def smart_termination(msg):
    content = msg.get("content", "")
    # 檢查多種終止標記
    terminate_keywords = ["TERMINATE", "完成", "結束", "任務完成"]
    return any(keyword in content for keyword in terminate_keywords)

agent = autogen.AssistantAgent(
    name="assistant",
    llm_config=llm_config,
    is_termination_msg=smart_termination
)
```

### 問題 2：代碼執行失敗

**症狀**：
- `CodeExecutor` 報錯
- 代碼無法運行
- Docker 相關錯誤

**原因**：
- 工作目錄不存在
- Docker 未安裝或未運行
- 缺少依賴包
- 權限問題

**解決方案**：

```python
import os

# ✅ 方案 1：確保工作目錄存在
work_dir = "coding_workspace"
os.makedirs(work_dir, exist_ok=True)

executor = autogen.UserProxyAgent(
    name="executor",
    code_execution_config={
        "work_dir": work_dir,
        "use_docker": False,
    }
)

# ✅ 方案 2：禁用 Docker（如果未安裝）
code_execution_config = {
    "work_dir": "workspace",
    "use_docker": False,  # 不使用 Docker
}

# ✅ 方案 3：檢查 Docker 是否可用
import subprocess

def is_docker_available():
    try:
        subprocess.run(["docker", "--version"],
                      capture_output=True,
                      check=True)
        return True
    except:
        return False

use_docker = is_docker_available()
code_execution_config = {
    "work_dir": "workspace",
    "use_docker": use_docker,
}

# ✅ 方案 4：添加錯誤處理
try:
    executor.initiate_chat(assistant, message=task)
except Exception as e:
    print(f"執行失敗: {e}")
    # 降級到不使用代碼執行
    executor.code_execution_config = False
    executor.initiate_chat(assistant, message=task)
```

```python
def chat_with_retry(agent, message, max_retries=3):
    """帶重試的對話"""
    for attempt in range(max_retries):
        try:
            return agent.initiate_chat(assistant, message=message)
        except Exception as e:
            if "rate_limit" in str(e).lower():
                wait_time = (2 ** attempt) * 10  # 指數退避
                print(f"達到速率限制，等待 {wait_time} 秒...")
                time.sleep(wait_time)
            else:
                raise
    raise Exception("達到最大重試次數")
```

### 問題 4：內存不足

**症狀**：
- `MemoryError`
- 程序運行變慢
- 系統卡頓

**原因**：
- 對話歷史過長
- 沒有清理舊消息
- 並發 Agent 過多

**解決方案**：

```python
# ✅ 方案 1：限制對話歷史長度
agent = autogen.AssistantAgent(
    name="assistant",
    llm_config=llm_config,
    max_consecutive_auto_reply=5  # 限制回合數
)

# ✅ 方案 2：定期清理歷史
def clear_old_messages(agent, keep_last_n=10):
    """保留最近 N 條消息"""
    for other_agent in agent.chat_messages:
        messages = agent.chat_messages[other_agent]
        if len(messages) > keep_last_n:
            agent.chat_messages[other_agent] = messages[-keep_last_n:]

# 使用
clear_old_messages(agent, keep_last_n=20)

# ✅ 方案 3：使用消息摘要
code_execution_config = {
    "work_dir": "workspace",
    "use_docker": False,
    "last_n_messages": 3,  # 只傳遞最近 3 條消息給代碼執行器
}

# ✅ 方案 4：批量處理
def process_tasks_in_batches(tasks, agent, batch_size=5):
    """分批處理任務"""
    results = []
    for i in range(0, len(tasks), batch_size):
        batch = tasks[i:i + batch_size]
        for task in batch:
            result = agent.initiate_chat(assistant, message=task)
            results.append(result)
        # 清理歷史
        clear_old_messages(agent)
    return results
```

### 問題 5：群組對話混亂

**症狀**：
- Agent 發言順序不合理
- 某些 Agent 從不發言
- 對話跑題

**原因**：
- 沒有設置發言順序
- system_message 不清晰
- 沒有發言邏輯

**解決方案**：

```python
# ✅ 方案 1：使用輪流發言
def round_robin_speaker_selection(last_speaker, groupchat):
    """輪流選擇發言者"""
    agents = groupchat.agents
    try:
        current_idx = agents.index(last_speaker)
        next_idx = (current_idx + 1) % len(agents)
        return agents[next_idx]
    except:
        return agents[0]

groupchat = autogen.GroupChat(
    agents=[agent1, agent2, agent3],
    messages=[],
    max_round=10,
    speaker_selection_method=round_robin_speaker_selection
)

# ✅ 方案 2：基於角色的發言順序
def role_based_selection(last_speaker, groupchat):
    """根據角色決定下一個發言者"""
    role_sequence = {
        "ProductManager": "Architect",
        "Architect": "Developer",
        "Developer": "QA",
        "QA": "ProductManager"
    }

    next_role = role_sequence.get(last_speaker.name)
    for agent in groupchat.agents:
        if agent.name == next_role:
            return agent
    return groupchat.agents[0]

# ✅ 方案 3：清晰的 system_message
product_manager_msg = """你是產品經理。

你的職責：
1. 理解需求
2. 設計功能
3. 協調團隊

發言規則：
- 首先發言，闡述需求
- 聽取其他角色意見
- 最後確認驗收

發言格式：
[產品經理] 內容...
"""
```

## 成本控制

### 1. Token 使用監控

```python
class TokenTracker:
    """Token 使用追蹤器"""

    def __init__(self):
        self.total_tokens = 0
        self.total_cost = 0
        self.pricing = {
            "gpt-4": {"input": 0.03, "output": 0.06},  # per 1K tokens
            "gpt-3.5-turbo": {"input": 0.0015, "output": 0.002},
        }

    def estimate_tokens(self, text):
        """估算 token 數（簡化版）"""
        return len(text) // 4

    def track_usage(self, model, input_text, output_text):
        """追蹤使用情況"""
        input_tokens = self.estimate_tokens(input_text)
        output_tokens = self.estimate_tokens(output_text)

        self.total_tokens += input_tokens + output_tokens

        if model in self.pricing:
            cost = (
                (input_tokens / 1000) * self.pricing[model]["input"] +
                (output_tokens / 1000) * self.pricing[model]["output"]
            )
            self.total_cost += cost

    def get_summary(self):
        """獲取使用摘要"""
        return {
            "total_tokens": self.total_tokens,
            "total_cost": f"${self.total_cost:.4f}"
        }

# 使用
tracker = TokenTracker()
# ... 在對話中追蹤 ...
print(tracker.get_summary())
```

```python
import time
from functools import wraps

def timing_decorator(func):
    """計時裝飾器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} 執行時間: {end - start:.2f} 秒")
        return result
    return wrapper

# 使用
@timing_decorator
def run_chat(agent, message):
    return agent.initiate_chat(assistant, message=message)
```

### 3. 監控和告警

```python
class Monitor:
    """監控系統"""

    def __init__(self):
        self.metrics = {
            "total_requests": 0,
            "failed_requests": 0,
            "total_tokens": 0,
            "total_cost": 0,
        }

    def record_request(self, success: bool, tokens: int = 0, cost: float = 0):
        """記錄請求"""
        self.metrics["total_requests"] += 1
        if not success:
            self.metrics["failed_requests"] += 1
        self.metrics["total_tokens"] += tokens
        self.metrics["total_cost"] += cost

        # 檢查告警條件
        self.check_alerts()

    def check_alerts(self):
        """檢查告警"""
        # 錯誤率告警
        if self.metrics["total_requests"] > 0:
            error_rate = self.metrics["failed_requests"] / self.metrics["total_requests"]
            if error_rate > 0.1:  # 10% 錯誤率
                self.send_alert(f"高錯誤率: {error_rate:.2%}")

        # 成本告警
        if self.metrics["total_cost"] > 100:  # $100
            self.send_alert(f"成本超標: ${self.metrics['total_cost']:.2f}")

    def send_alert(self, message: str):
        """發送告警"""
        logger.error(f"ALERT: {message}")
        # 這裡可以集成 Slack、郵件等告警方式
```
