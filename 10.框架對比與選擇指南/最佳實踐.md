### 1. 模組化設計

**原則**: 將系統分解為獨立的可重用組件

```python
# ❌ 不好的做法：所有邏輯混在一起
def process_query(query):
    # 加載數據
    documents = load_documents()
    # 創建索引
    index = create_index(documents)
    # 查詢
    result = query_index(index, query)
    # 後處理
    return post_process(result)

# ✅ 好的做法：模組化
class DocumentLoader:
    def load(self, source): ...

class IndexManager:
    def create_index(self, documents): ...
    def query(self, query): ...

class QueryProcessor:
    def __init__(self, loader, index_manager):
        self.loader = loader
        self.index_manager = index_manager

    def process(self, query):
        documents = self.loader.load()
        index = self.index_manager.create_index(documents)
        return self.index_manager.query(query)
```

### 1. 分塊策略優化

#### 固定大小分塊 vs 語義分塊

```python
# ❌ 簡單固定大小分塊 - 可能切斷語義
from langchain.text_splitter import CharacterTextSplitter

splitter = CharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
chunks = splitter.split_text(text)

# ✅ 語義感知分塊 - 保持語義完整性
from langchain.text_splitter import RecursiveCharacterTextSplitter

splitter = RecursiveCharacterTextSplitter(
    chunk_size=1000,
    chunk_overlap=200,
    separators=["\n\n", "\n", ". ", " ", ""],  # 優先在段落/句子邊界分割
    length_function=len,
)
chunks = splitter.split_text(text)

# ✅✅ 更進階: 基於句子的語義分塊
from llama_index.node_parser import SentenceSplitter

parser = SentenceSplitter(
    chunk_size=1024,
    chunk_overlap=20,
    paragraph_separator="\n\n",
    secondary_chunking_regex="[^,.;。]+[,.;。]?"
)
nodes = parser.get_nodes_from_documents(documents)
```

#### 針對不同文檔類型的分塊策略

```python
def get_optimal_splitter(doc_type: str):
    """根據文檔類型返回最佳分塊器"""

    if doc_type == "code":
        return RecursiveCharacterTextSplitter(
            chunk_size=1500,
            chunk_overlap=200,
            separators=["\nclass ", "\ndef ", "\n\n", "\n", " "]
        )

    elif doc_type == "markdown":
        return RecursiveCharacterTextSplitter(
            chunk_size=1000,
            chunk_overlap=100,
            separators=["\n## ", "\n### ", "\n\n", "\n", " "]
        )

    elif doc_type == "legal":
        # 法律文檔需要保持更大的上下文
        return RecursiveCharacterTextSplitter(
            chunk_size=2000,
            chunk_overlap=400,
            separators=["\n\n", "\n", ". ", " "]
        )

    else:  # 一般文章
        return RecursiveCharacterTextSplitter(
            chunk_size=1000,
            chunk_overlap=150
        )
```

### 3. 重排序 (Reranking)

檢索後重排序可以顯著提升結果質量:

```python
from langchain.retrievers import ContextualCompressionRetriever
from langchain.retrievers.document_compressors import CohereRerank

# 基礎檢索器
base_retriever = vector_store.as_retriever(search_kwargs={"k": 20})

# 使用 Cohere Rerank 重排序
compressor = CohereRerank(top_n=5)  # 從20個結果中選出最相關的5個

compression_retriever = ContextualCompressionRetriever(
    base_compressor=compressor,
    base_retriever=base_retriever
)

# ✅ 更好的方式: 使用跨編碼器重排序
from sentence_transformers import CrossEncoder

class CrossEncoderReranker:
    def __init__(self, model_name="cross-encoder/ms-marco-MiniLM-L-6-v2"):
        self.model = CrossEncoder(model_name)

    def rerank(self, query: str, documents: list, top_k: int = 5):
        # 計算查詢和每個文檔的相關性分數
        pairs = [[query, doc.page_content] for doc in documents]
        scores = self.model.predict(pairs)

        # 按分數排序
        sorted_docs = [doc for _, doc in sorted(
            zip(scores, documents),
            key=lambda x: x[0],
            reverse=True
        )]

        return sorted_docs[:top_k]

# 使用
reranker = CrossEncoderReranker()
initial_docs = base_retriever.get_relevant_documents(query, k=20)
reranked_docs = reranker.rerank(query, initial_docs, top_k=5)
```

### 4. 查詢改寫與擴展

改善用戶查詢以獲得更好的檢索結果:

```python
from langchain.chains import LLMChain
from langchain.prompts import PromptTemplate

# 查詢擴展 - 生成多個相關查詢
query_expansion_template = """
給定用戶查詢,生成3個語義相似但表達不同的查詢變體:

原始查詢: {query}

變體查詢 (每行一個):
"""

expansion_chain = LLMChain(
    llm=llm,
    prompt=PromptTemplate.from_template(query_expansion_template)
)

def expand_query(query: str):
    """擴展查詢為多個變體"""
    result = expansion_chain.run(query=query)
    queries = [q.strip() for q in result.split('\n') if q.strip()]
    return [query] + queries  # 包含原始查詢

# HyDE (Hypothetical Document Embeddings)
hyde_template = """
基於以下問題,生成一個假設性的詳細回答:

問題: {query}

假設性回答:
"""

hyde_chain = LLMChain(
    llm=llm,
    prompt=PromptTemplate.from_template(hyde_template)
)

def hyde_retrieval(query: str, retriever):
    """使用 HyDE 方法檢索"""
    # 生成假設性文檔
    hypothetical_doc = hyde_chain.run(query=query)

    # 用假設性文檔檢索而非原始查詢
    docs = retriever.get_relevant_documents(hypothetical_doc)

    return docs
```

```python
from langchain.vectorstores import Chroma

# ✅ 自引用檢索 - 先檢索元數據,再檢索內容
class SelfQueryRetriever:
    def __init__(self, vectorstore, llm):
        self.vectorstore = vectorstore
        self.llm = llm

    def retrieve(self, query: str):
        # 使用 LLM 從查詢中提取過濾條件
        filter_prompt = f"""
        從以下查詢中提取結構化的過濾條件:
        查詢: {query}

            query,
            filter=filters
```

### 2. 緩存策略

**原則**: 緩存昂貴的操作結果

```python
from functools import lru_cache
import hashlib

class CachedEmbeddings:
    def __init__(self, embeddings_model, cache_dir="./cache"):
        self.model = embeddings_model
        self.cache_dir = cache_dir

    def get_embedding(self, text):
        # 生成緩存鍵
        cache_key = hashlib.md5(text.encode()).hexdigest()
        cache_path = f"{self.cache_dir}/{cache_key}.npy"

        # 檢查緩存
        if os.path.exists(cache_path):
            return np.load(cache_path)

        # 生成並緩存
        embedding = self.model.embed_query(text)
        np.save(cache_path, embedding)
        return embedding
```

### 2. 模型選擇策略

```python
class AdaptiveModelSelector:
    """根據任務複雜度選擇模型"""

    def select_model(self, task_complexity):
        if task_complexity == "simple":
            return "gpt-3.5-turbo"  # 便宜
        elif task_complexity == "medium":
            return "gpt-4-turbo"  # 平衡
        else:
            return "gpt-4"  # 強大但貴

    def estimate_complexity(self, query):
        # 簡單啟發式規則
        if len(query) < 50 and "?" in query:
            return "simple"
        elif any(keyword in query for keyword in ["分析", "總結", "解釋"]):
            return "medium"
        else:
```

```python
from langgraph.graph import StateGraph, END
from langchain.chat_models import ChatOpenAI
from langchain.memory import RedisChatMessageHistory

# 定義狀態
class CustomerServiceState(TypedDict):
    messages: list
    user_intent: str
    resolved: bool
    escalate: bool

# 定義節點
def classify_intent(state):
    """分類用戶意圖"""
    llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)
    # 使用便宜的模型進行意圖分類
    intent = llm.predict(f"Classify intent: {state['messages'][-1]}")
    return {"user_intent": intent}

def handle_simple_query(state):
    """處理簡單查詢"""
    if state["user_intent"] in ["product_info", "shipping"]:
        # 從知識庫檢索
        response = retriever.get_relevant_documents(state["messages"][-1])
        return {"resolved": True}
    return {"resolved": False}

def handle_complex_query(state):
    """處理複雜查詢"""
    llm = ChatOpenAI(model="gpt-4-turbo", temperature=0.7)
    # 使用強大模型處理複雜問題
    response = llm.predict(state["messages"][-1])
    return {"messages": state["messages"] + [response]}

def should_escalate(state):
    """判斷是否需要人工介入"""
    if state["user_intent"] in ["complaint", "refund"] or not state["resolved"]:
        return {"escalate": True}
    return {"escalate": False}

# 構建流程圖
workflow = StateGraph(CustomerServiceState)

workflow.add_node("classify", classify_intent)
workflow.add_node("simple", handle_simple_query)
workflow.add_node("complex", handle_complex_query)
workflow.add_node("escalate", should_escalate)

# 添加邊
workflow.set_entry_point("classify")
workflow.add_conditional_edges(
    "classify",
    lambda x: "simple" if x["user_intent"] in ["product_info", "shipping"] else "complex"
)
workflow.add_edge("simple", "escalate")
workflow.add_edge("complex", "escalate")
workflow.add_conditional_edges(
    "escalate",
    lambda x: END if not x["escalate"] else "human_handoff"
)

# 編譯
app = workflow.compile()

# 使用 Redis 持久化對話歷史
def chat(user_id: str, message: str):
    history = RedisChatMessageHistory(
        session_id=user_id,
        url="redis://localhost:6379"
    )

    result = app.invoke({
        "messages": history.messages + [message],
        "user_intent": "",
        "resolved": False,
        "escalate": False
    })

    history.add_user_message(message)
    history.add_ai_message(result["messages"][-1])

    return result
```
