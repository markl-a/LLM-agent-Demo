## 🏗️ 架構設計原則

### 1. 模組化設計

**原則**: 將系統分解為獨立的可重用組件

```python
# ❌ 不好的做法：所有邏輯混在一起
def process_query(query):
    # 加載數據
    documents = load_documents()
    # 創建索引
    index = create_index(documents)
    # 查詢
    result = query_index(index, query)
    # 後處理
    return post_process(result)

# ✅ 好的做法：模組化
class DocumentLoader:
    def load(self, source): ...

class IndexManager:
    def create_index(self, documents): ...
    def query(self, query): ...

class QueryProcessor:
    def __init__(self, loader, index_manager):
        self.loader = loader
        self.index_manager = index_manager

    def process(self, query):
        documents = self.loader.load()
        index = self.index_manager.create_index(documents)
        return self.index_manager.query(query)
```

### 1. 批處理

**問題**: 逐個處理效率低
**解決**: 批量處理

```python
# ❌ 逐個處理
for text in texts:
    embedding = get_embedding(text)  # 每次調用 API
    store_embedding(embedding)

# ✅ 批量處理
batch_size = 100
for i in range(0, len(texts), batch_size):
    batch = texts[i:i+batch_size]
    embeddings = get_embeddings_batch(batch)  # 一次調用
    store_embeddings(embeddings)
```

### 2. 緩存策略

**原則**: 緩存昂貴的操作結果

```python
from functools import lru_cache
import hashlib

class CachedEmbeddings:
    def __init__(self, embeddings_model, cache_dir="./cache"):
        self.model = embeddings_model
        self.cache_dir = cache_dir

    def get_embedding(self, text):
        # 生成緩存鍵
        cache_key = hashlib.md5(text.encode()).hexdigest()
        cache_path = f"{self.cache_dir}/{cache_key}.npy"

        # 檢查緩存
        if os.path.exists(cache_path):
            return np.load(cache_path)

        # 生成並緩存
        embedding = self.model.embed_query(text)
        np.save(cache_path, embedding)
        return embedding
```

```python
import asyncio
from langchain.llms import ChatOpenAI

# ✅ 異步處理多個請求
async def process_queries(queries):
    llm = ChatOpenAI()
    tasks = [llm.agenerate([query]) for query in queries]
    results = await asyncio.gather(*tasks)
    return results
```

```python
# ✅ 針對不同類型的文檔使用不同的分塊策略
def get_splitter(doc_type):
    if doc_type == "code":
```

```python
    def select_model(self, task_complexity):
        if task_complexity == "simple":
            return "gpt-3.5-turbo"  # 便宜
        elif task_complexity == "medium":
            return "gpt-4-turbo"  # 平衡

            return "simple"
        elif any(keyword in query for keyword in ["分析", "總結", "解釋"]):
            return "medium"
        else:
            return "complex"
```

```python
def validate_query(query: str) -> bool:
    """驗證用戶輸入"""
    if not query or len(query.strip()) == 0:
        raise ValueError("查詢不能為空")

    if len(query) > 10000:
        raise ValueError("查詢太長")

    # 檢查惡意輸入
    if any(pattern in query.lower() for pattern in ["ignore previous", "system:"]):
        raise ValueError("檢測到潛在的注入攻擊")

    return True
```

### 2. Prompt 注入防護

```python
def sanitize_input(user_input: str) -> str:
    """清理用戶輸入，防止 prompt 注入"""

    # 移除系統指令關鍵字
    dangerous_patterns = [
        "ignore previous instructions",
        "disregard all",
        "forget everything",
        "system:",
        "assistant:",
    ]

    cleaned = user_input
    for pattern in dangerous_patterns:
        cleaned = cleaned.replace(pattern, "")

    return cleaned

# 使用
safe_input = sanitize_input(user_input)
response = llm.predict(safe_input)
```

```python
def filter_sensitive_info(text: str) -> str:
    """過濾敏感信息"""
    import re

    # 遮蔽電子郵件
    text = re.sub(r'\b[\w.-]+@[\w.-]+\.\w+\b', '[EMAIL]', text)

    # 遮蔽電話號碼
    text = re.sub(r'\b\d{3}-\d{4}-\d{4}\b', '[PHONE]', text)

    # 遮蔽信用卡號
    text = re.sub(r'\b\d{4}-\d{4}-\d{4}-\d{4}\b', '[CREDIT_CARD]', text)

    return text
```

### 4. 訪問控制

```python
class SecureRAGSystem:
    def __init__(self, rag_system, auth_service):
        self.rag = rag_system
        self.auth = auth_service

    def query(self, question, user_id, api_key):
        # 驗證用戶
        if not self.auth.verify_user(user_id, api_key):
            raise PermissionError("未授權訪問")

        # 檢查配額
        if not self.auth.check_quota(user_id):
            raise QuotaExceededError("已達使用限額")

        # 執行查詢
        result = self.rag.query(question)

        # 記錄使用
        self.auth.log_usage(user_id, len(question))

        return result
```

```python
class HealthChecker:
    def __init__(self, rag_system):
        self.rag = rag_system

    def check_health(self):
        checks = {
            "llm": self._check_llm(),
            "vector_store": self._check_vector_store(),
            "memory": self._check_memory()
        }

        all_healthy = all(checks.values())

        return {
            "status": "healthy" if all_healthy else "unhealthy",
            "checks": checks
        }

    def _check_llm(self):
        try:
            self.rag.llm.predict("test")
            return True
        except:
            return False

    def _check_vector_store(self):
        try:
            self.rag.vector_store.similarity_search("test", k=1)
            return True
        except:
            return False

    def _check_memory(self):
        import psutil
        memory_percent = psutil.virtual_memory().percent
        return memory_percent < 90
```
