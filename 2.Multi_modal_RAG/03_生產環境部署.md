### 組件職責

#### 1. 負載均衡層
- **職責**：流量分發、SSL 終止、DDoS 防護
- **技術選型**：
  - Nginx（自建）
  - AWS ALB / GCP Load Balancer（雲端）
  - Cloudflare（CDN + 安全）

#### 2. API Gateway
- **職責**：請求驗證、限流、路由
- **技術選型**：FastAPI / Django / Flask

#### 3. RAG Worker
- **職責**：處理 RAG 查詢、文檔處理
- **技術選型**：
  - Celery（傳統任務隊列）
  - Ray（分佈式計算）
  - AWS Lambda（Serverless）

#### 4. 數據層
- **向量庫**：Pinecone / Milvus / Qdrant
- **文檔存儲**：Redis / PostgreSQL / S3
- **快取**：Redis / Memcached

## 基礎設施選型

### 雲端服務商比較

| 服務 | AWS | GCP | Azure | 阿里雲 |
|------|-----|-----|-------|--------|
| 計算 | EC2, Lambda | Compute Engine, Cloud Run | VM, Functions | ECS, Function Compute |
| 存儲 | S3 | Cloud Storage | Blob Storage | OSS |
| 數據庫 | RDS, DynamoDB | Cloud SQL, Firestore | SQL Database | RDS, TableStore |
| 向量庫 | - | - | - | - |
| 容器 | ECS, EKS | GKE | AKS | ACK |
| 監控 | CloudWatch | Cloud Monitoring | Azure Monitor | CloudMonitor |

**推薦組合**：
- **小規模**（< 1000 QPS）：單雲，使用託管服務
- **中規模**（1000-10000 QPS）：單雲，Kubernetes
- **大規模**（> 10000 QPS）：多雲，自建+託管混合

# 決策樹
def select_vector_database(requirements):
    """
    選擇向量數據庫

    requirements = {
        "scale": "small" | "medium" | "large",
        "budget": "low" | "medium" | "high",
        "deployment": "cloud" | "on-premise" | "hybrid"
    }
    """

    if requirements["scale"] == "small":
        if requirements["deployment"] == "on-premise":
            return "Chroma"  # 輕量、易部署
        else:
            return "Pinecone"  # 全託管、免運維

        if requirements["budget"] == "low":

            return "Milvus"  # 開源、可擴展

#### 3. API 服務實現

```python
# api/main.py
from fastapi import FastAPI, HTTPException, Depends
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
import os
from rag_service import RAGService
from prometheus_client import Counter, Histogram, generate_latest
import time

app = FastAPI(title="Multimodal RAG API")

# CORS
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# Prometheus 指標
REQUEST_COUNT = Counter(
    'rag_requests_total',
    'Total RAG requests',
    ['endpoint', 'status']
)

REQUEST_LATENCY = Histogram(
    'rag_request_latency_seconds',
    'RAG request latency',
    ['endpoint']
)

# 初始化 RAG 服務
rag_service = RAGService(
    openai_api_key=os.getenv("OPENAI_API_KEY"),
    pinecone_api_key=os.getenv("PINECONE_API_KEY"),
    redis_url=os.getenv("REDIS_URL")
)

class QueryRequest(BaseModel):
    query: str
    max_results: int = 5
    include_images: bool = True

class QueryResponse(BaseModel):
    answer: str
    sources: list
    latency: float

@app.post("/query", response_model=QueryResponse)
async def query_rag(request: QueryRequest):
    """RAG 查詢接口"""
    start_time = time.time()

    try:
        # 執行 RAG 查詢
        result = await rag_service.query(
            query=request.query,
            max_results=request.max_results,
            include_images=request.include_images
        )

        latency = time.time() - start_time

        # 記錄指標
        REQUEST_COUNT.labels(endpoint='query', status='success').inc()
        REQUEST_LATENCY.labels(endpoint='query').observe(latency)

        return QueryResponse(
            answer=result["answer"],
            sources=result["sources"],
            latency=latency
        )

    except Exception as e:
        REQUEST_COUNT.labels(endpoint='query', status='error').inc()
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/health")
async def health_check():
    """健康檢查"""
    return {
        "status": "healthy",
        "service": "multimodal-rag-api"
    }

@app.get("/metrics")
async def metrics():
    """Prometheus 指標端點"""
    return generate_latest()

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

## 災難恢復

### 備份策略

```python
# backup.py
import boto3
from datetime import datetime
import json

class BackupManager:
    """備份管理器"""

    def __init__(self, s3_bucket):
        self.s3 = boto3.client('s3')
        self.bucket = s3_bucket

    def backup_vector_index(self, pinecone_index):
        """備份向量索引（元數據）"""

        # Pinecone 不支持直接備份，保存索引配置
        metadata = {
            "name": pinecone_index.name,
            "dimension": pinecone_index.dimension,
            "metric": pinecone_index.metric,
            "num_vectors": pinecone_index.describe_index_stats()["total_vector_count"],
            "timestamp": datetime.utcnow().isoformat()
        }

        # 上傳到 S3
        self.s3.put_object(
            Bucket=self.bucket,
            Key=f"backups/vector_index_{datetime.utcnow().strftime('%Y%m%d_%H%M%S')}.json",
            Body=json.dumps(metadata)
        )

    def backup_documents(self, redis_client):
        """備份文檔存儲"""

        # 導出所有文檔
        docs = {}
        for key in redis_client.scan_iter("doc:*"):
            docs[key.decode()] = redis_client.get(key).decode()

        # 上傳到 S3
        self.s3.put_object(
            Bucket=self.bucket,
            Key=f"backups/documents_{datetime.utcnow().strftime('%Y%m%d_%H%M%S')}.json",
            Body=json.dumps(docs)
        )

    def restore_documents(self, backup_file, redis_client):
        """恢復文檔"""

        # 從 S3 下載
        obj = self.s3.get_object(Bucket=self.bucket, Key=backup_file)
        docs = json.loads(obj['Body'].read())

        # 恢復到 Redis
        pipe = redis_client.pipeline()
        for key, value in docs.items():
            pipe.set(key, value)
        pipe.execute()
```

### 自動備份腳本

```bash
#!/bin/bash
# backup.sh

set -e

BACKUP_DIR="/backups"
DATE=$(date +%Y%m%d_%H%M%S)

echo "開始備份：$DATE"

# 1. 備份 Redis
echo "備份 Redis..."
redis-cli --rdb $BACKUP_DIR/redis_$DATE.rdb

# 2. 備份配置
echo "備份配置文件..."
tar -czf $BACKUP_DIR/config_$DATE.tar.gz /app/config

# 3. 上傳到 S3

import boto3
from botocore.exceptions import ClientError

class SecretManager:
    """密鑰管理器"""

    def __init__(self):
        self.client = boto3.client('secretsmanager')

    def get_secret(self, secret_name):
        """獲取密鑰"""
        try:
            response = self.client.get_secret_value(SecretId=secret_name)
            return response['SecretString']
        except ClientError as e:
            raise Exception(f"無法獲取密鑰：{e}")

    def rotate_secret(self, secret_name, new_value):
        """輪換密鑰"""
        self.client.put_secret_value(
            SecretId=secret_name,
            SecretString=new_value
        )

# 使用

from pydantic import BaseModel, validator
import re

class QueryRequest(BaseModel):
    query: str
    max_results: int = 5

    @validator('query')
    def validate_query(cls, v):
        # 長度限制
        if len(v) > 1000:
            raise ValueError('查詢太長，最多 1000 字符')

        # 防止注入攻擊
        if re.search(r'<script|javascript:|onerror=', v, re.I):
            raise ValueError('查詢包含非法內容')

        return v

    @validator('max_results')
    def validate_max_results(cls, v):
        if v < 1 or v > 20:
            raise ValueError('max_results 必須在 1-20 之間')
        return v
```
